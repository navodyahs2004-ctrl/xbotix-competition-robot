import serial
import time

# --- Configuration ---
# Update 'COM_PORT' to the correct port for your Arduino Nano
SERIAL_PORT = '/dev/ttyACM0' 
BAUD_RATE = 9600

# Commands to send to the Arduino
CMD_STOP = b'1'   # Simulates GPIO_state = 1 (Stop movement)
CMD_START = b'0'  # Simulates GPIO_state = 0 (Start movement)

def send_command(command: bytes):
    """Sends a command to the Arduino via the serial connection."""
    try:
        # Initialize serial connection
        ser = serial.Serial(SERIAL_PORT, BAUD_RATE, timeout=1)
        time.sleep(2) # Wait for the connection to initialize

        print(f"Sending command: {command.decode()} to Arduino...")
        ser.write(command)
        print("Command sent successfully.")

        ser.close()
    except serial.SerialException as e:
        print(f"Error connecting or communicating with Arduino: {e}")
    except Exception as e:
        print(f"An unexpected error occurred: {e}")

def stop_robot():
    """Simulates GPIO_state = 1: Stops the movement script."""
    send_command(CMD_STOP)

def start_robot():
    """Simulates GPIO_state = 0: Starts/Resumes the movement script."""
    send_command(CMD_START)

if __name__ == '__main__':
    # --- Example Usage ---
    print("--- Testing Robot Control ---")
    start_robot()
    time.sleep(5)
    stop_robot()
    time.sleep(3)
    start_robot()
